function TechnologyNode({
  technology,
  occasionalLabel,
}: {
  technology: Technology;
  occasionalLabel: string;
}) {
  return (
    <div
      title={
        technology.occasional
          ? `${technology.name} — ${occasionalLabel}`
          : technology.name
      }
      className="
        group/node relative flex min-w-0
        items-center gap-2.5 rounded-lg
        border border-white/[0.06]
        bg-white/[0.025] px-3 py-2.5
        transition-all duration-200
        hover:border-white/20
        hover:bg-white/[0.07]
      "
    >
      <span
        aria-hidden="true"
        className="
          flex h-8 w-8 shrink-0
          items-center justify-center
          rounded-md font-mono
          text-xs font-bold
          transition-transform
          motion-safe:group-hover/node:scale-110
        "
        style={{
          color: technology.color,
          backgroundColor: `${technology.color}14`,
        }}
      >
        {technology.symbol}
      </span>

      <span
        className="
          text-xs font-medium
          leading-5 text-slate-300
          group-hover/node:text-white
        "
      >
        {technology.name}
      </span>

      {technology.occasional && (
        <span
          className="
            absolute right-1.5 top-1.5
            h-1.5 w-1.5 rounded-full
            bg-amber-400
          "
        />
      )}
    </div>
  );
}

function Connector({
  label,
  active,
}: {
  label: string;
  active: boolean;
}) {
  return (
    <div
      className="
        relative flex h-12
        items-center justify-center
      "
    >
      <div
        aria-hidden="true"
        className="
          absolute inset-y-0 left-1/2
          w-px -translate-x-1/2
        "
        style={{
          background: active
            ? 'linear-gradient(transparent, #38bdf8, transparent)'
            : 'linear-gradient(transparent, #334155, transparent)',
        }}
      />

      <div
        aria-hidden="true"
        className="
          architecture-packet
          absolute left-1/2 top-0
          h-2 w-2 -translate-x-1/2
          rounded-full bg-sky-400
        "
        style={{
          opacity: active ? 1 : 0.45,
        }}
      />

      <span
        className="
          relative z-10 flex items-center
          gap-2 rounded-full border
          border-white/[0.07]
          bg-[#0c111b] px-3 py-1
          font-mono text-[10px]
          text-slate-400
        "
      >
        <ArrowDown size={11} />

        {label}
      </span>
    </div>
  );
}

function ArchitectureLayer({
  layer,
  copy,
  active,
  dimmed,
  onSelect,
}: {
  layer: Layer;
  copy: Copy;
  active: boolean;
  dimmed: boolean;
  onSelect: () => void;
}) {
  const Icon = layer.icon;

  const id = layer.id as LayerId;

  const description =
    copy[`${id}Description` as const];

  return (
    <article
      className="
        relative overflow-hidden
        rounded-xl border
        transition-all duration-300
      "
      style={{
        borderColor: active
          ? `${layer.color}65`
          : 'rgba(148,163,184,0.13)',

        backgroundColor: active
          ? `${layer.color}08`
          : 'rgba(15,23,42,0.45)',

        opacity: dimmed ? 0.55 : 1,

        boxShadow: active
          ? `0 0 30px ${layer.color}09`
          : 'none',
      }}
    >
      <div
        aria-hidden="true"
        className="absolute inset-y-0 left-0 w-[2px]"
        style={{
          backgroundColor: layer.color,
          opacity: active ? 1 : 0.55,
        }}
      />

      <button
        type="button"
        onClick={onSelect}
        aria-pressed={active}
        className="
          flex w-full items-center
          gap-3 px-5 py-4
          text-left
          transition-colors
          hover:bg-white/[0.025]
        "
      >
        <span
          className="
            flex h-10 w-10 shrink-0
            items-center justify-center
            rounded-lg
          "
          style={{
            color: layer.color,
            backgroundColor: `${layer.color}12`,
          }}
        >
          <Icon size={21} strokeWidth={1.6} />
        </span>

        <span className="min-w-0 flex-1">
          <span
            className="
              block text-sm font-semibold
              tracking-tight text-slate-100
            "
          >
            {copy[id]}
          </span>

          <span
            className="
              mt-1 block text-xs
              leading-5 text-slate-400
            "
          >
            {description}
          </span>
        </span>

        <span
          aria-hidden="true"
          className="h-2 w-2 shrink-0 rounded-full"
          style={{
            backgroundColor: layer.color,
            boxShadow: active
              ? `0 0 12px ${layer.color}`
              : 'none',
          }}
        />
      </button>

      <div
        className="
          grid grid-cols-2 gap-2
          px-5 pb-5
          sm:grid-cols-3
          xl:grid-cols-4
        "
      >
        {layer.technologies.map((technology) => (
          <TechnologyNode
            key={technology.name}
            technology={technology}
            occasionalLabel={copy.occasional}
          />
        ))}
      </div>
    </article>
  );
}

              "

            "

      >

          "
        >

          </span>

          <span className="flex items-center gap-1.5">
            <Workflow size={12} />
            Systems
          </span>
        </div>

        <span
          aria-live="polite"

          {activeLayer
            ? `${copy.selected}: ${copy[activeLayer]}`

        </span>
